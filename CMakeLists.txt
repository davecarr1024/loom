cmake_minimum_required(VERSION 3.20)
project(loom LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(ENABLE_COVERAGE "Instrument production tests" OFF)
add_library(loom INTERFACE)
target_include_directories(loom INTERFACE include)
target_compile_options(loom INTERFACE -Wall -Wextra -Wpedantic)
include(FetchContent)
FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git GIT_TAG v1.14.0)
FetchContent_MakeAvailable(googletest)
enable_testing()
add_executable(circuit_tests tests/circuit_test.cpp tests/not_test.cpp)
target_link_libraries(circuit_tests PRIVATE loom GTest::gtest_main)
if(ENABLE_COVERAGE)
  target_compile_options(circuit_tests PRIVATE --coverage -O0 -g)
  target_link_options(circuit_tests PRIVATE --coverage)
endif()
include(GoogleTest)
gtest_discover_tests(circuit_tests)
add_executable(transfer examples/transfer.cpp)
target_link_libraries(transfer PRIVATE loom)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME wrong_width COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/compile_fail.py ${CMAKE_CXX_COMPILER} ${CMAKE_SOURCE_DIR})

add_executable(not_demo examples/not.cpp)
target_link_libraries(not_demo PRIVATE loom)
foreach(case logic_width logic_role numeric_bit custom_atom)
  add_test(NAME ${case} COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/compile_fail.py ${CMAKE_CXX_COMPILER} ${CMAKE_SOURCE_DIR} ${case})
endforeach()
